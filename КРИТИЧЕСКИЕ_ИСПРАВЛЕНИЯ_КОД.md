## 1. JSON.parse без try-catch - ИСПРАВИТЬ ПЕРВЫМ

### Где находится проблема:

#### 📍 frontend/src/services/api.js:132
```javascript
// СЕЙЧАС (ОПАСНО):
export const getCurrentUser = () => {
  const user = localStorage.getItem('user');
  return user ? JSON.parse(user) : null;
};
```

**Что произойдет:**
- localStorage содержит поврежденные данные
- JSON.parse выбрасывает exception
- Весь app крашнется с белым экраном
- Пользователь не может зайти в приложение

**Как исправить:**
```javascript
export const getCurrentUser = () => {
  try {
    const user = localStorage.getItem('user');
    return user ? JSON.parse(user) : null;
  } catch (error) {
    console.warn('Failed to parse user from localStorage:', error);
    // Очищаем поврежденные данные
    localStorage.removeItem('user');
    return null;
  }
};
```

## 3. Python subprocess утечка

### 📍 backend/src/controllers/morphologyController.js:25

**СЕЙЧАС (ОПАСНО):**
```javascript
const proc = spawn('python3', ['-u', analyzerPath]);

// Если proc.stdin.write() бросит exception ДО timeout
// то proc.kill() может не вызваться
// → процесс-зомби

const timeoutId = setTimeout(() => {
  proc.kill('SIGKILL');
}, 10000);
```

**КАК ИСПРАВИТЬ:**
```javascript
const morphologyController = {
  analyze: async (req, res) => {
    let proc = null;
    let timeoutId = null;
    
    try {
      proc = spawn('python3', ['-u', analyzerPath]);
      
      // Set timeout
      timeoutId = setTimeout(() => {
        if (proc && !proc.killed) {
          console.warn('Morphology analyzer timeout, killing process');
          proc.kill('SIGKILL');
        }
      }, 10000);
      
      let output = '';
      let errorOutput = '';
      
      proc.stdout.on('data', (data) => {
        output += data.toString();
      });
      
      proc.stderr.on('data', (data) => {
        errorOutput += data.toString();
      });
      
      proc.on('close', (code) => {
        clearTimeout(timeoutId);
        
        if (code !== 0) {
          return res.status(500).json({ 
            message: 'Analyzer failed',
            error: errorOutput 
          });
        }
        
        try {
          const result = JSON.parse(output);
          res.json(result);
        } catch (error) {
          res.status(500).json({ message: 'Failed to parse analyzer output' });
        }
      });
      
      // Write input
      proc.stdin.write(JSON.stringify({ text: req.body.text }));
      proc.stdin.end();
      
    } catch (error) {
      console.error('Morphology analyzer error:', error);
      res.status(500).json({ message: 'Analyzer error' });
    } finally {
      // ВСЕГДА чистим таймаут и убиваем процесс
      if (timeoutId) {
        clearTimeout(timeoutId);
      }
      if (proc && !proc.killed) {
        proc.kill('SIGKILL');
      }
    }
  }
};
```

```javascript
  }

    message={error}
    onRetry={fetchTasks}
  />
);
```

## 5. Race condition в AuthContext

### 📍 frontend/src/contexts/AuthContext.jsx:14

**СЕЙЧАС (ОПАСНО):**
```javascript
useEffect(() => {
  validateAuth(); // async функция
}, []);

// Но loading используется неправильно:
// children рендерятся ДО завершения validateAuth
```

**ЧТО ПРОИСХОДИТ:**
1. Пользователь обновляет страницу
2. validateAuth() запускается (async)
3. loading=true НО children уже рендерятся
4. Компоненты делают API calls с невалидным токеном
5. Получают 401
6. Редирект на login
7. Потеря несохраненных данных

**КАК ИСПРАВИТЬ:**

```javascript
// В AuthContext.jsx:
const [loading, setLoading] = useState(true); // ← ВАЖНО: начинаем с true

useEffect(() => {
  const validate = async () => {
    try {
      await validateAuth();
    } finally {
      setLoading(false); // ← устанавливаем false ТОЛЬКО после завершения
    }
  };
  
  validate();
}, []);
```

```javascript
// В ProtectedRoute компоненте:
const ProtectedRoute = ({ children }) => {
  const { user, loading } = useAuth();
  
  // ВАЖНО: показываем loader пока идет проверка
  if (loading) {
    return (
      <div className="loading-screen">
        <LoadingSpinner />
        <p>Проверка авторизации...</p>
      </div>
    );
  }
  
  if (!user) {
    return <Navigate to="/login" replace />;
  }
  
  return children;
};
```

## 6. JWT в localStorage - XSS уязвимость

### Текущая схема (НЕБЕЗОПАСНО):
```
Frontend → localStorage.setItem('token', jwt)
→ Любой XSS скрипт может украсть: localStorage.getItem('token')
```

### ✅ БЕЗОПАСНАЯ схема - httpOnly cookie

#### Backend изменения:

```javascript
// backend/src/controllers/authController.js
const login = async (req, res) => {
  // ... проверка пароля ...
  
  const token = jwt.sign({ userId: user.id }, JWT_SECRET, { expiresIn: '7d' });
  
  // Отправить токен в httpOnly cookie
  res.cookie('token', token, {
    httpOnly: true,        // ← JavaScript не может прочитать
    secure: true,          // ← только HTTPS
    sameSite: 'strict',    // ← защита от CSRF
    maxAge: 7 * 24 * 60 * 60 * 1000 // 7 дней
  });
  
  // НЕ отправлять токен в JSON response
  res.json({ 
    user: { 
      id: user.id, 
      username: user.username,
      email: user.email 
    } 
  });
};

const logout = async (req, res) => {
  res.clearCookie('token');
  res.json({ message: 'Logged out' });
};
```

```javascript
// backend/src/middleware/auth.js
const authenticateToken = (req, res, next) => {
  // Читать токен из cookie вместо Authorization header
  const token = req.cookies.token;
  
  if (!token) {
    return res.status(401).json({ message: 'Unauthorized' });
  }
  
  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ message: 'Invalid token' });
    }
    req.user = user;
    next();
  });
};
```

#### Frontend изменения:

